cmake_minimum_required(VERSION 3.20)
project(asstats LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASSTATS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(asstats src/decoder.cpp src/pcap.cpp)
target_include_directories(asstats PUBLIC include)
target_compile_options(asstats PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ASSTATS_SANITIZERS)
  target_compile_options(asstats PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(asstats PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nf9-replay src/replay.cpp)
target_link_libraries(nf9-replay PRIVATE asstats)
target_compile_options(nf9-replay PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
enable_testing()
add_executable(decoder-tests tests/decoder_tests.cpp)
target_link_libraries(decoder-tests PRIVATE asstats)
add_test(NAME decoder-tests COMMAND decoder-tests)
add_executable(pcap-tests tests/pcap_tests.cpp)
target_link_libraries(pcap-tests PRIVATE asstats)
add_test(NAME pcap-tests COMMAND pcap-tests)

find_package(Threads REQUIRED)
add_library(asstats-live src/config.cpp src/app_config.cpp src/queue.cpp src/udp.cpp src/async_output.cpp src/delivery.cpp src/aggregation_config.cpp src/aggregation.cpp)
target_link_libraries(asstats-live PUBLIC asstats Threads::Threads)
target_compile_options(asstats-live PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(nf9-receiver src/receiver.cpp)
target_link_libraries(nf9-receiver PRIVATE asstats-live)
target_compile_options(nf9-receiver PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(live-tests tests/live_tests.cpp)
target_link_libraries(live-tests PRIVATE asstats-live)
add_test(NAME live-unit COMMAND live-tests)
add_test(NAME udp-sockets COMMAND live-tests --sockets)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME receiver-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/receiver_integration.py $<TARGET_FILE:nf9-receiver>)
set_tests_properties(receiver-integration PROPERTIES TIMEOUT 30)

add_executable(nf9-aggregate-replay src/aggregate_replay.cpp)
target_link_libraries(nf9-aggregate-replay PRIVATE asstats-live)
add_executable(aggregation-tests tests/aggregation_tests.cpp)
target_link_libraries(aggregation-tests PRIVATE asstats-live)
add_test(NAME aggregation-unit COMMAND aggregation-tests)
add_test(NAME output-failures COMMAND aggregation-tests --output)
set_tests_properties(output-failures PROPERTIES TIMEOUT 10)
add_test(NAME slow-diagnostic COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/slow_diagnostic.py $<TARGET_FILE:nf9-receiver>)
set_tests_properties(slow-diagnostic PROPERTIES TIMEOUT 15)
add_executable(app-config-tests tests/app_config_tests.cpp)
target_link_libraries(app-config-tests PRIVATE asstats-live)
add_test(NAME app-config COMMAND app-config-tests)
add_test(NAME config-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/config_integration.py $<TARGET_FILE:nf9-receiver>)
set_tests_properties(config-integration PROPERTIES TIMEOUT 15)

find_package(CURL REQUIRED)
add_executable(nf9-delivery-worker src/delivery_worker.cpp)
target_link_libraries(nf9-delivery-worker PRIVATE asstats-live CURL::libcurl)
target_compile_options(nf9-delivery-worker PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_dependencies(nf9-receiver nf9-delivery-worker)
add_executable(delivery-driver tests/delivery_driver.cpp)
target_link_libraries(delivery-driver PRIVATE asstats-live)
add_dependencies(delivery-driver nf9-delivery-worker)
add_test(NAME delivery-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/delivery_integration.py $<TARGET_FILE:delivery-driver>)
set_tests_properties(delivery-integration PROPERTIES TIMEOUT 30)
